NewStud: Группы и студенты для предмета

Процедура am_GetStudentsAndGroupsForSubject_v1

  • Описание
  • Код процедуры
  • Как вызывать

Назначение: получить список групп и студентов для конкретного предмета в заданном учебном плане, курсе и семестре, c учётом «живых» статусов студентов и реальной ситуации с подпланами (точное соответствие подплану или fallback на общие группы запуска плана).

Алгоритм

  • Вычислить Num_Term = (Course - 1) * 2 + Semester и Year_Enter = Year_Study - (Course - 1).
  • Найти запуск плана Plan_Year_Enter для (@Nfirm, @PlanID, @Year_Enter).
  • Получить множество подпланов предмета в семестре из Comp_Term (валидность по Beg_Act/End_Act).
  • Подобрать группы запуска плана: сначала по найденным подпланам (режим by_subplan), если пусто — взять общие группы запуска (FK_Sub_Plan IS NULL) — режим fallback_common.
  • Вернуть студентов из выбранных групп, у которых последний статус в {1,2,3,4,5,9}.

Возвращаемые поля (сокращённо)

  • Группы PrK_Group_Stud, Group_Name, FK_Sub_Plan, Sub_Plan_Name, FK_Pлан_Year, Owner
  • Студенты PrK_Entrant, ФИО из Person
  • Статус Last_Status_Type, Last_Status_Name
  • Тех. поле Source_Mode = by_subplan | fallback_common
  • Контекст Plan_ID, Subject_ID

Примечание: процедура специально реализует «умный» fallback на общие группы, когда фактические группы набора ещё не размечены по подпланам, хотя предмет в плане привязан к подплану.

Код процедуры

CREATE OR ALTER PROC dbo.am_GetStudentsAndGroupsForSubject_v1
  @Nfirm      INT,   -- факультет
  @PlanID     INT,   -- Educ_Plan.PrK_Plan
  @SubjectID  INT,   -- Subject_Name.PrK_Subject_Name
  @Year_Study INT,   -- год начала учебного года (например, 2025 для 2025/26)
  @Course     INT,   -- курс (1..)
  @Semester   INT    -- 1 = осень, 2 = весна
AS
BEGIN
  SET NOCOUNT ON;

  IF @Course < 1 OR @Semester NOT IN (1,2)
  BEGIN
    -- Некорректные параметры: вернём пустой набор
    SELECT TOP (0)
         CAST(NULL AS INT)  AS Plan_ID,
         CAST(NULL AS INT)  AS Subject_ID,
         CAST(NULL AS INT)  AS PrK_Group_Stud,
         CAST(NULL AS INT)  AS Owner,
         CAST(NULL AS NVARCHAR(40)) AS Group_Name,
         CAST(NULL AS INT)  AS FK_Sub_Plan,
         CAST(NULL AS NVARCHAR(200)) AS Sub_Plan_Name,
         CAST(NULL AS INT)  AS FK_Plan_Year,
         CAST(NULL AS INT)  AS PrK_Entrant,
         CAST(NULL AS NVARCHAR(60)) AS Family,
         CAST(NULL AS NVARCHAR(60)) AS First_Name,
         CAST(NULL AS NVARCHAR(60)) AS Second_Name,
         CAST(NULL AS INT)  AS Last_Status_Type,
         CAST(NULL AS NVARCHAR(60)) AS Last_Status_Name,
         CAST(NULL AS NVARCHAR(20)) AS Source_Mode;
    RETURN;
  END;

  DECLARE @Num_Term   INT = (@Course - 1) * 2 + @Semester;  -- 1..12
  DECLARE @Year_Enter INT = @Year_Study - (@Course - 1);

  DECLARE @PYE INT;
  SELECT @PYE = pye.PrK_Plan_Year
  FROM dbo.Plan_Year_Enter pye
  WHERE pye.NFirm = @Nfirm
    AND pye.FK_Plan = @PlanID
    AND pye.Year_Enter = @Year_Enter;

  IF @PYE IS NULL
  BEGIN
    -- План не запускался в этот набор — пусто
    SELECT TOP (0)
         CAST(NULL AS INT)  AS Plan_ID,
         CAST(NULL AS INT)  AS Subject_ID,
         CAST(NULL AS INT)  AS PrK_Group_Stud,
         CAST(NULL AS INT)  AS Owner,
         CAST(NULL AS NVARCHAR(40)) AS Group_Name,
         CAST(NULL AS INT)  AS FK_Sub_Plan,
         CAST(NULL AS NVARCHAR(200)) AS Sub_Plan_Name,
         CAST(NULL AS INT)  AS FK_Plan_Year,
         CAST(NULL AS INT)  AS PrK_Entrant,
         CAST(NULL AS NVARCHAR(60)) AS Family,
         CAST(NULL AS NVARCHAR(60)) AS First_Name,
         CAST(NULL AS NVARCHAR(60)) AS Second_Name,
         CAST(NULL AS INT)  AS Last_Status_Type,
         CAST(NULL AS NVARCHAR(60)) AS Last_Status_Name,
         CAST(NULL AS NVARCHAR(20)) AS Source_Mode;
    RETURN;
  END;

  -- ПодPlanы предмета в данном семестре (валидные на год набора)
  IF OBJECT_ID('tempdb..#Subplans') IS NOT NULL DROP TABLE #Subplans;
  SELECT DISTINCT ct.FK_Sub_Plan
  INTO #Subplans
  FROM dbo.Subject_Name_Order sno
  JOIN dbo.Comp_Term ct
    ON ct.NFirm      = sno.NFirm
   AND ct.CodOrderID = sno.PrK_Subject_Order
  WHERE sno.NFirm = @Nfirm
    AND sno.FK_Plan = @PlanID
    AND sno.FK_Subject_Name = @SubjectID
    AND ct.Num_Term = @Num_Term
    AND ct.Beg_Act <= @Year_Enter
    AND (ct.End_Act = 0 OR ct.End_Act >= @Year_Enter);

  -- Кандидаты групп
  IF OBJECT_ID('tempdb..#Groups') IS NOT NULL DROP TABLE #Groups;
  CREATE TABLE #Groups (
    Nfirm           INT NOT NULL,
    PrK_Group_Stud  INT NOT NULL,
    Owner           INT NOT NULL,
    FK_Sub_Plan     INT NULL,
    FK_Plan_Year    INT NOT NULL,
    Group_Name      NVARCHAR(40) NOT NULL,
    Source_Mode     NVARCHAR(20) NOT NULL  -- by_subplan | fallback_common
  );

  -- 1) Точный режим: группы запуска по найденным подPlanам
  INSERT INTO #Groups (Nfirm, PrK_Group_Stud, Owner, FK_Sub_Plan, FK_Plan_Year, Group_Name, Source_Mode)
  SELECT gs.NFirm, gs.PrK_Group_Stud, gs.Owner, gs.FK_Sub_Plan, gs.FK_Plan_Year, gs.Group_Name, N'by_subplan'
  FROM dbo.Group_Stud gs
  WHERE gs.NFirm = @Nfirm
    AND gs.FK_Plan_Year = @PYE
    AND gs.FK_Sub_Plan IS NOT NULL
    AND EXISTS (SELECT 1 FROM #Subplans s WHERE s.FK_Sub_Plan = gs.FK_Sub_Plan);

  -- 2) Fallback: если по подPlanам пусто, берём общие группы запуска
  IF NOT EXISTS (SELECT 1 FROM #Groups)
  BEGIN
    INSERT INTO #Groups (Nfirm, PrK_Group_Stud, Owner, FK_Sub_Plan, FK_Plan_Year, Group_Name, Source_Mode)
    SELECT gs.NFirm, gs.PrK_Group_Stud, gs.Owner, gs.FK_Sub_Plan, gs.FK_Plan_Year, gs.Group_Name, N'fallback_common'
    FROM dbo.Group_Stud gs
    WHERE gs.NFirm = @Nfirm
      AND gs.FK_Plan_Year = @PYE
      AND gs.FK_Sub_Plan IS NULL;
  END;

  -- Последний «живой» статус по студенту
  ;WITH LastStatus AS (
    SELECT s.FK_Entrant,
         s.FK_Status_Type,
         st.Status_Type_Name,
         ROW_NUMBER() OVER (
           PARTITION BY s.FK_Entrant
           ORDER BY s.Status_Date DESC, s.PrK_Status DESC
         ) AS rn
    FROM dbo.Status s
    JOIN dbo.Status_Type st ON st.PrK_Status_Type = s.FK_Status_Type
    WHERE s.NFirm = @Nfirm
  )
  SELECT
    @PlanID                   AS Plan_ID,
    @SubjectID                AS Subject_ID,
    g.PrK_Group_Stud,
    g.Owner,
    g.Group_Name,
    g.FK_Sub_Plan,
    sp.Sub_Plan_Name,
    g.FK_Plan_Year,
    e.PrK_Entrant,
    p.Family,
    p.First_Name,
    p.Second_Name,
    ls.FK_Status_Type     AS Last_Status_Type,
    ls.Status_Type_Name   AS Last_Status_Name,
    g.Source_Mode
  FROM #Groups g
  JOIN dbo.Entrant e
    ON e.NFirm = g.NFirm
   AND e.FK_Group_Stud = g.PrK_Group_Stud
   AND e.Owner_Group   = g.Owner
  JOIN LastStatus ls
    ON ls.FK_Entrant = e.PrK_Entrant
   AND ls.rn = 1
   AND ls.FK_Status_Type IN (1,2,3,4,5,9)  -- «живые» статусы
  JOIN dbo.Person p
    ON p.PrK_Person = e.FK_Person
  LEFT JOIN dbo.Sub_Plan sp
    ON sp.NFirm = g.NFirm
   AND sp.PrK_Sub_Plan = g.FK_Sub_Plan
  ORDER BY g.Source_Mode, g.Group_Name, p.Family, p.First_Name, p.Second_Name;
END

Как вызывать

EXEC dbo.am_GetStudentsAndGroupsForSubject_v1
   @Nfirm      = 1,
   @PlanID     = 671,
   @SubjectID  = 155,   -- например «Алгебра»
   @Year_Study = 2025,  -- учебный год 2025/26
   @Course     = 1,
   @Semester   = 1;

Ожидаемая логика

  • Если в Group_Stud ещё нет групп, размеченных по нужным подпланам предмета, будут строки с Source_Mode = 'fallback_common' — это означает, что подобраны общие группы запуска плана.
  • Как только фактическая привязка групп к подпланам появится, вернутся строки с Source_Mode = 'by_subplan'.
