Как получить даты начала и конца семестра

NewStud / Sched: границы терма из Graph + Num_Week + Duration

Идея

Даты начала/конца семестра в модели NewStud корректнее брать из Graph (календарь плана) + Num_Week (стартовая неделя) + Duration (длительность в неделях).

Ниже — готовая процедура (SQL Server 2008+; без TRY_CONVERT), возвращает StartDate и EndDate (включительно) для выбранного плана, учебного года и терма (Graph.Term).

Как «приклеить» год

Поле Num_Week.Date_Week хранится как dd.mm без года, поэтому год нужно вычислять по учебному году:

  • если месяц >= 9 → год = @Year_Study (например, 2025 для 2025/26);
  • иначе → год = @Year_Study + 1.

Процедура am_GetTermDates

IF OBJECT_ID('dbo.am_GetTermDates', 'P') IS NOT NULL
    DROP PROC dbo.am_GetTermDates;
GO

CREATE PROC dbo.am_GetTermDates
    @PlanID     INT,  -- Educ_Plan.PrK_Plan
    @Year_Study INT,  -- год начала учебного года (напр., 2025 для 2025/26)
    @Term       INT   -- сквозной номер терма (Graph.Term)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @Nfirm INT;
    SELECT @Nfirm = ep.Nfirm
    FROM dbo.Educ_Plan ep
    WHERE ep.PrK_Plan = @PlanID;

    IF @Nfirm IS NULL
    BEGIN
        RAISERROR(N'План не найден', 16, 1);
        RETURN;
    END

    /*
      Берём именно этап "Теоретическое обучение".
      Если у вас другой код этапа, поменяйте условие:
      - либо по FK_Educ_Stage = 1,
      - либо по названию Educ_Stage_Name LIKE N'Теорет%'.
    */
    ;WITH G AS (
        SELECT TOP (1)
            g.FK_Plan,
            g.Term,
            g.Course,
            g.Duration,
            g.FK_Week,
            nw.Date_Week,
            MonthNum = CAST(SUBSTRING(nw.Date_Week, 4, 2) AS INT)
        FROM dbo.Graph g
        JOIN dbo.Num_Week nw ON nw.PrK_Week = g.FK_Week
        WHERE g.FK_Plan = @PlanID
          AND g.NFirm   = @Nfirm
          AND g.Term    = @Term
          AND g.FK_Educ_Stage = 1
        ORDER BY g.Num_Stage
    )
    SELECT
        FK_Plan        = @PlanID,
        NFirm          = @Nfirm,
        Term           = g.Term,
        Course         = g.Course,
        DurationWeeks  = g.Duration,
        FK_Week        = g.FK_Week,
        Date_Week      = g.Date_Week,
        StartDate      = CONVERT(date,
                          g.Date_Week + '.' +
                          CAST(CASE WHEN g.MonthNum >= 9 THEN @Year_Study ELSE @Year_Study + 1 END AS varchar(4)),
                          104),
        -- Конец семестра как последний день последней недели (включительно)
        EndDate        = CONVERT(date,
                          DATEADD(day, 7 * g.Duration - 1,
                            CONVERT(datetime,
                              g.Date_Week + '.' +
                              CAST(CASE WHEN g.MonthNum >= 9 THEN @Year_Study ELSE @Year_Study + 1 END AS varchar(4)),
                              104)))
    FROM G g;
END
GO

Пример вызова

EXEC dbo.am_GetTermDates @PlanID = 671, @Year_Study = 2025, @Term = 1;  -- осень 1 курса
EXEC dbo.am_GetTermDates @PlanID = 671, @Year_Study = 2025, @Term = 2;  -- весна 1 курса

Если код стадии другой

Иногда «теоретическое обучение» имеет другой код. Тогда замените фильтр:

AND g.FK_Educ_Stage = 1

на вариант с именем стадии:

JOIN dbo.Educ_Stage es ON es.PrK_Educ_Stage = g.FK_Educ_Stage
...
AND es.Educ_Stage_Name LIKE N'Теорет%'

Почему Graph, а не Comp_Term

  • Graph задаёт календарное окно семестра (когда реально можно ставить занятия).
  • Comp_Term задаёт нагрузку по предметам внутри семестра (Quan_Week, типы занятий).

Обычно Graph.Duration и Comp_Term.Quan_Week должны совпадать для соответствующего терма (на этапе “теория”), но границы периода логичнее брать именно из Graph.

Проверка / диагностика

Если процедура возвращает корректные даты, но дальше «не бьётся» нагрузка/предметы, первым делом проверьте консистентность Graph.Duration vs Comp_Term.Quan_Week для того же Term/Num_Term и нужного набора.

Сводка: сколько недель по Graph и по Comp_Term

DECLARE
    @PlanID     INT = 671,
    @Year_Study INT = 2025,
    @Term       INT = 1;

DECLARE @Nfirm INT;
SELECT @Nfirm = ep.Nfirm
FROM dbo.Educ_Plan ep
WHERE ep.PrK_Plan = @PlanID;

;WITH G AS (
    SELECT TOP (1)
        g.FK_Plan,
        g.NFirm,
        g.Term,
        g.Course,
        g.Duration
    FROM dbo.Graph g
    WHERE g.FK_Plan = @PlanID
      AND g.NFirm   = @Nfirm
      AND g.Term    = @Term
      AND g.FK_Educ_Stage = 1
    ORDER BY g.Num_Stage
), Ctx AS (
    SELECT
        g.*,
        Year_Enter = @Year_Study - (g.Course - 1)
    FROM G g
)
SELECT
    c.FK_Plan,
    c.NFirm,
    c.Term,
    c.Course,
    Year_Enter  = c.Year_Enter,
    GraphWeeks  = c.Duration,
    CTItems     = COUNT(ct.PrK_Comp_Term),
    DistinctCTWeeks = COUNT(DISTINCT ct.Quan_Week),
    MinCTWeeks  = MIN(ct.Quan_Week),
    MaxCTWeeks  = MAX(ct.Quan_Week)
FROM Ctx c
LEFT JOIN dbo.Comp_Term ct
  ON ct.NFirm    = c.NFirm
 AND ct.Num_Term = c.Term
 AND ct.Beg_Act <= c.Year_Enter
 AND (ct.End_Act = 0 OR ct.End_Act >= c.Year_Enter)
LEFT JOIN dbo.Subject_Name_Order sno
  ON sno.PrK_Subject_Order = ct.CodOrderID
 AND sno.NFirm             = ct.NFirm
 -- Если в вашей схеме есть FK на план, включите жёсткую привязку:
 -- AND sno.FK_Plan           = c.FK_Plan
GROUP BY c.FK_Plan, c.NFirm, c.Term, c.Course, c.Year_Enter, c.Duration;

Как читать результат: если DistinctCTWeeks > 1 или диапазон MinCTWeeks..MaxCTWeeks не равен GraphWeeks, значит в Comp_Term по этому терму есть неоднородные компоненты или «съехали» данные.

Детализация: какие именно значения Quan_Week встречаются

DECLARE
    @PlanID     INT = 671,
    @Year_Study INT = 2025,
    @Term       INT = 1;

DECLARE @Nfirm INT;
SELECT @Nfirm = ep.Nfirm
FROM dbo.Educ_Plan ep
WHERE ep.PrK_Plan = @PlanID;

DECLARE @Course INT;
SELECT TOP (1) @Course = g.Course
FROM dbo.Graph g
WHERE g.FK_Plan = @PlanID
  AND g.NFirm   = @Nfirm
  AND g.Term    = @Term
  AND g.FK_Educ_Stage = 1
ORDER BY g.Num_Stage;

DECLARE @Year_Enter INT = @Year_Study - (@Course - 1);

SELECT
    ct.Quan_Week,
    Items = COUNT(*),
    Subjects = COUNT(DISTINCT ct.CodOrderID)
FROM dbo.Comp_Term ct
WHERE ct.NFirm    = @Nfirm
  AND ct.Num_Term = @Term
  AND ct.Beg_Act <= @Year_Enter
  AND (ct.End_Act = 0 OR ct.End_Act >= @Year_Enter)
GROUP BY ct.Quan_Week
ORDER BY ct.Quan_Week;
